import base64
import io
import numpy as np
import matplotlib.pyplot as plt
from PIL import Image
from google.colab import output

_canvas_store = {}

from PIL import Image, ImageFilter 

def to_mnist(img: np.ndarray) -> np.ndarray:
    ys, xs = np.where(img > 0.05) # Чуть снизили порог, чтобы не терять слабые касания
    if len(xs) == 0:
        return np.zeros((28, 28), dtype=np.float32)
        
    x0, x1 = xs.min(), xs.max() + 1
    y0, y1 = ys.min(), ys.max() + 1
    digit = img[y0:y1, x0:x1]
    h, w = digit.shape
    
    # 1. MNIST цифры занимают примерно 20 пикселей по большей стороне
    scale = 20.0 / max(h, w)
    new_w = max(1, int(round(w * scale)))
    new_h = max(1, int(round(h * scale)))
    
    pil = Image.fromarray((digit * 255).astype(np.uint8), mode='L')
    pil = pil.resize((new_w, new_h), Image.LANCZOS)
    
    # --- НОВЫЙ БЛОК: ИМИТАЦИЯ ТОЛЩИНЫ И РАЗМЫТИЯ MNIST ---
    # Небольшое размытие сглаживает углы холста
    pil = pil.filter(ImageFilter.GaussianBlur(radius=0.6))
    
    resized = np.asarray(pil, dtype=np.float32) / 255.0
    
    # Искусственно делаем линии чуть жирнее (повышаем контраст полутонов)
    resized = np.power(resized, 0.7) 
    # -----------------------------------------------------
    
    out = np.zeros((28, 28), dtype=np.float32)
    ox = (28 - new_w) // 2
    oy = (28 - new_h) // 2
    out[oy:oy + new_h, ox:ox + new_w] = resized
    
    # Центрирование по массе (оставляем как было, оно у вас отличное)
    total = out.sum()
    if total > 0:
        yy, xx = np.indices(out.shape)
        cy = (out * yy).sum() / total
        cx = (out * xx).sum() / total
        out = np.roll(out, int(round(13.5 - cy)), axis=0)
        out = np.roll(out, int(round(13.5 - cx)), axis=1)
        
    return np.clip(out, 0.0, 1.0)


def _keras_preprocess(mnist, mode):
    x = mnist.astype(np.float32)
    if mode == 'div255':   return x
    if mode == 'raw':      return x
    if mode == 'pt_norm':  return (x - 0.1307) / 0.3081
    if mode == 'x255':     return x * 255.0
    if mode == 'invert':   return 1.0 - x
    raise ValueError(mode)


# ---------- ОБНОВЛЕННЫЙ ПРИЕМ ПАЙПЛАЙНА ----------
def _receive_canvas(b64, preprocess='div255'):
    try:
        # Декодируем бинарный PNG из Base64
        img_bytes = base64.b64decode(b64)
        pil_img = Image.open(io.BytesIO(img_bytes)).convert('RGB')
        rgba = np.array(pil_img)
        
        # Берем первый канал (R), так как фон черный, а рисуем белым
        img = rgba[:, :, 0].astype(np.float32) / 255.0
        mnist = to_mnist(img)
        
        _canvas_store['img']   = img
        _canvas_store['mnist'] = mnist
        err_msg = None
    except Exception as e:
        print(f"Ошибка декодирования: {e}")
        return

    probs, pred = None, None
    try:
        x = _keras_preprocess(mnist, preprocess)

        # Проверяем форму входа модели
        in_shape = model.input_shape
        if len(in_shape) == 4:
            inp = x.reshape(1, 28, 28, 1)
        elif len(in_shape) == 2:
            inp = x.reshape(1, -1)
        else:
            raise ValueError(f"Неожиданная форма входа модели: {in_shape}")

        probs = model.predict(inp, verbose=0)[0]
        pred = int(np.argmax(probs))
    except Exception as e:
        err_msg = f"{type(e).__name__}: {e}"

    # ---------- Визуализация (остается вашей) ----------
    fig = plt.figure(figsize=(9, 3.2))
    gs  = fig.add_gridspec(1, 3, width_ratios=[1, 1, 2.2])

    ax0 = fig.add_subplot(gs[0, 0])
    ax0.imshow(img, cmap='gray', vmin=0, vmax=1); ax0.axis('off')
    ax0.set_title('Холст', fontsize=10)

    ax1 = fig.add_subplot(gs[0, 1])
    ax1.imshow(mnist, cmap='gray', vmin=0, vmax=1); ax1.axis('off')
    ax1.set_title('MNIST 28×28', fontsize=10)

    ax2 = fig.add_subplot(gs[0, 2])
    if probs is not None:
        colors = ['#cccccc'] * 10
        colors[pred] = '#2ca02c'
        ax2.barh(range(10), probs, color=colors)
        ax2.set_yticks(range(10))
        ax2.set_yticklabels([str(d) for d in range(10)], fontsize=9)
        ax2.invert_yaxis()
        ax2.set_xlim(0, 1)
        ax2.set_xlabel('вероятность', fontsize=9)
        ax2.set_title(f'Предсказание: {pred}  ({probs[pred]:.1%})',
                      fontsize=11, fontweight='bold')
        for d, p in enumerate(probs):
            if p >= 0.01:
                ax2.text(min(p + 0.02, 0.98), d, f'{p:.0%}',
                         va='center', fontsize=8,
                         color='#1a7a1a' if d == pred else 'black')
        ax2.grid(axis='x', alpha=0.3)
    else:
        ax2.text(0.5, 0.5, err_msg or 'модель недоступна',
                 ha='center', va='center', fontsize=9, color='#a00')
        ax2.axis('off')

    plt.tight_layout()
    plt.show()

    if probs is not None:
        print(f"Нарисовано: {pred}   уверенность: {probs[pred]:.2%}")
        print("  " + "  ".join(f"{d}:{probs[d]:.2f}" for d in range(10)))
    else:
        print(f"Предсказание недоступно: {err_msg}")

# ---------- Регистрация ----------
output.register_callback('_receive_canvas', lambda b64: _receive_canvas(b64, preprocess='x255'))
